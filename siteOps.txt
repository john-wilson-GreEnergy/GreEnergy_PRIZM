17:function findArraysByObjectKeys(obj: any, requiredKeys: string[], results: any[] = []) {
18:    if (!obj || typeof obj !== 'object') return results;
19:    if (Array.isArray(obj)) {
20:        if (obj.length > 0 && typeof obj[0] === 'object' && requiredKeys.every(k => k in obj[0])) {
21:            results.push(...obj);
22:        } else {
23:            obj.forEach(o => findArraysByObjectKeys(o, requiredKeys, results));
24:        }
25:    } else {
26:        for (const [k, v] of Object.entries(obj)) {
27:            findArraysByObjectKeys(v, requiredKeys, results);
28:        }
29:    }
30:    return results;
31:}
